-- 1. Criação do Banco de Dados
CREATE DATABASE IF NOT EXISTS Projeto_Final;
USE Projeto_Final;

-- --------------------------------------------------------
-- 2. Tabela de Usuários (Com CEP adicionado)
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS Usuarios (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    DataNascimento DATETIME DEFAULT CURRENT_TIMESTAMP,
    CPF VARCHAR(14) NOT NULL,
    Email VARCHAR(120),
    Telefone VARCHAR(20),
    
    -- Endereço
    CEP VARCHAR(10),       -- <--- CAMPO NOVO ADICIONADO
    Cidade VARCHAR(50),
    Estado VARCHAR(50),
    Rua VARCHAR(50),
    Bairro VARCHAR(50),
    Numero VARCHAR(10),
    
    -- Login e Acesso
    Senha VARCHAR(30),
    Cargo ENUM('Gerente','Vendedor','Estoquista'),
    
    UNIQUE INDEX cpf_UNIQUE (CPF ASC)
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Clientes (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    CPF VARCHAR(14) NOT NULL UNIQUE,
    Telefone VARCHAR(20),
    CEP VARCHAR(10),
    Cidade VARCHAR(50),
    Estado VARCHAR(50),
    Rua VARCHAR(50),
    Bairro VARCHAR(50),
    Numero VARCHAR(10)
) ENGINE = InnoDB;

-- --------------------------------------------------------
-- 3. Tabela de Produtos
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS Produtos (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    
    -- Dados Básicos
    Nome VARCHAR(150) NOT NULL,
    Cor VARCHAR(50),
    Categoria VARCHAR(50),
    
    -- Estoque
    QtdAtual INT DEFAULT 0,
    QtdMax INT DEFAULT 0,
    QtdMin INT DEFAULT 0,
    
    -- Financeiro
    Valor DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    
    -- Identificação
    CodigoBarras VARCHAR(50) UNIQUE NOT NULL,
    
    -- Imagem
    Foto LONGBLOB
) ENGINE = InnoDB;

-- --------------------------------------------------------
-- 4. Tabela de PDV_Caixa (Fluxo de Caixa)
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS PDV_Caixa (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    DataAbertura DATETIME DEFAULT NOW(),
    DataFechamento DATETIME NULL,
    ValorInicial DECIMAL(10, 2), 
    ValorFinal DECIMAL(10, 2) NULL,
    UsuarioResponsavel VARCHAR(100), 
    GerenteLiberacao VARCHAR(100) NULL, 
    StatusCaixa VARCHAR(20) DEFAULT 'Aberto'
) ENGINE = InnoDB;

-- --------------------------------------------------------
-- 5. Tabela de Vendas
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS Vendas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    Id_Cliente int not null,
    ID_FluxoCaixa INT NOT NULL,    
    data_venda DATETIME NOT NULL,
    valor_total DECIMAL(10,2),
    forma_pagamento VARCHAR(50),   
    
    foreign key(Id_Cliente) references Clientes(Id),
    FOREIGN KEY (id_usuario) REFERENCES Usuarios(id),
    FOREIGN KEY (ID_FluxoCaixa) REFERENCES PDV_Caixa(ID)
) ENGINE = InnoDB;

-- --------------------------------------------------------
-- 6. Tabela de Itens_Venda
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS Itens_Venda (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_venda INT NOT NULL,
    id_produto INT NOT NULL,
    quantidade INT NOT NULL,
    valor_unitario DECIMAL(10,2),
    subtotal DECIMAL(10,2),
    
    FOREIGN KEY (id_venda) REFERENCES Vendas(id),
    FOREIGN KEY (id_produto) REFERENCES Produtos(Id)
) ENGINE = InnoDB;

-- --------------------------------------------------------
-- 7. Tabela de Pagamentos_Venda
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS pagamentos_venda (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_venda INT NOT NULL,
    tipo_pagamento VARCHAR(50), 
    valor DECIMAL(10,2),
    
    FOREIGN KEY (id_venda) REFERENCES Vendas(id)
) ENGINE = InnoDB;



-- --------------------------------------------------------
-- 8. Tabela de Log do Sistema
-- --------------------------------------------------------
CREATE TABLE IF NOT EXISTS log_sistema (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome_usuario VARCHAR(100) NOT NULL,
    acao VARCHAR(150) NOT NULL,
    formulario VARCHAR(100),
    data_hora DATETIME,
    INDEX IX_UserLogs_Data (data_hora DESC)
) ENGINE = InnoDB;
