-- ==================================================================================
-- 1. TABELAS DE BASE E CADASTROS

CREATE TABLE Usuarios (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    DataNascimento DATETIME DEFAULT CURRENT_TIMESTAMP,
    CPF VARCHAR(14) NOT NULL UNIQUE,
    Email VARCHAR(120),
    Telefone VARCHAR(20),
    CEP VARCHAR(10),
    Cidade VARCHAR(50),
    Estado VARCHAR(50),
    Rua VARCHAR(50),
    Bairro VARCHAR(50),
    Numero VARCHAR(10),
    Senha VARCHAR(30),
    Cargo ENUM('Gerente','Vendedor','Estoquista')
) ENGINE = InnoDB;

CREATE TABLE Clientes (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    CPF VARCHAR(14) NOT NULL UNIQUE,
    Telefone VARCHAR(20),
    CEP VARCHAR(10),
    Cidade VARCHAR(50),
    Estado VARCHAR(50),
    Rua VARCHAR(50),
    Bairro VARCHAR(50),

) ENGINE = InnoDB;

CREATE TABLE Fornecedores (

    Estado VARCHAR(50),

    Bairro VARCHAR(50),

) ENGINE = InnoDB;

CREATE TABLE Grades (
    Id INT AUTO_INCREMENT PRIMARY KEY,

CREATE TABLE Categorias (
    Id INT AUTO_INCREMENT PRIMARY KEY,

CREATE TABLE Estilos (

-- ==================================================================================
-- 2. TABELAS COM DEPENDÊNCIAS DE NÍVEL 1
-- ==================================================================================

CREATE TABLE Fornecedores_Condicoes (
    FornecedorId INT PRIMARY KEY,
    PrazoEntregaDias INT NOT NULL DEFAULT 7,
    TaxaFretePadrao DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedores(Id) ON DELETE CASCADE
) ENGINE = InnoDB;

CREATE TABLE Fornecedores_Categorias (
    FornecedorId INT NOT NULL,
    CategoriaId INT NOT NULL,
    PRIMARY KEY (FornecedorId, CategoriaId),
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedores(Id) ON DELETE CASCADE,
    FOREIGN KEY (CategoriaId) REFERENCES Categorias(Id)
) ENGINE = InnoDB;

CREATE TABLE Grades_Tamanhos (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    GradeId INT NOT NULL,
    Tamanho VARCHAR(10) NOT NULL,
    OrdemExibicao INT NOT NULL,
    FOREIGN KEY (GradeId) REFERENCES Grades(Id) ON DELETE CASCADE
) ENGINE = InnoDB;

CREATE TABLE SubCategorias (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    CategoriaPaiId INT NOT NULL,
    Nome VARCHAR(50) NOT NULL,
    GradePadraoId INT NOT NULL,
    FOREIGN KEY (CategoriaPaiId) REFERENCES Categorias(Id),
    FOREIGN KEY (GradePadraoId) REFERENCES Grades(Id),
    UNIQUE KEY UK_Cat_Sub (CategoriaPaiId, Nome)
) ENGINE = InnoDB;

-- ==================================================================================
-- 3. CATÁLOGO DE PRODUTOS
-- ==================================================================================

CREATE TABLE Produtos (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(150) NOT NULL,
    Descricao TEXT,
    Categoria VARCHAR(50), 
    SubCategoria VARCHAR(50),
    Estilo VARCHAR(50),
    Colecao VARCHAR(50),
    Composicao VARCHAR(100),
    Modelagem VARCHAR(50),
    Cor VARCHAR(50),
    ValorCompra DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    ValorVendaBase DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    FotoCapa LONGBLOB,
    DataCadastro DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE = InnoDB;

CREATE TABLE Produtos_Variacoes (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    ProdutoId INT NOT NULL,
    Tamanho VARCHAR(10) NOT NULL,
    CodigoBarrasEAN VARCHAR(50) UNIQUE NULL,
    ValorVendaAtual DECIMAL(10, 2) NOT NULL DEFAULT 0.00,
    QtdAtual INT DEFAULT 0,
    QtdMin INT DEFAULT 0,
    QtdMax INT DEFAULT 0,
    FOREIGN KEY (ProdutoId) REFERENCES Produtos(Id) ON DELETE CASCADE
) ENGINE = InnoDB;

CREATE TABLE Produtos_Fornecedores_Catalogo (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    ProdutoPaiId INT NOT NULL,
    FornecedorId INT NOT NULL,
    PrecoCustoTabela DECIMAL(10, 2) NOT NULL,
    CodigoProdutoNoFornecedor VARCHAR(50),
    IsFornecedorPrincipal BOOLEAN DEFAULT FALSE,
    DataUltimaAtualizacao DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (ProdutoPaiId) REFERENCES Produtos(Id) ON DELETE CASCADE,
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedores(Id) ON DELETE CASCADE,
    UNIQUE KEY UK_Produto_Fornecedor (ProdutoPaiId, FornecedorId)
) ENGINE = InnoDB;    

-- ==================================================================================
-- 4. VENDAS E OPERACIONAL
-- ==================================================================================

CREATE TABLE Vendas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    Id_Cliente INT NULL,
    ID_FluxoCaixa INT NOT NULL,       
    data_venda DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    valor_total DECIMAL(10,2) NOT NULL,
    forma_pagamento VARCHAR(50),
    obs TEXT,
    FOREIGN KEY (Id_Cliente) REFERENCES Clientes(Id),
    FOREIGN KEY (id_usuario) REFERENCES Usuarios(id),
    FOREIGN KEY (ID_FluxoCaixa) REFERENCES PDV_Caixa(ID)
) ENGINE = InnoDB;

CREATE TABLE Itens_Venda (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_venda INT NOT NULL,
    id_variacao INT NOT NULL, 
    quantidade INT NOT NULL,
    valor_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2),
    FOREIGN KEY (id_venda) REFERENCES Vendas(id) ON DELETE CASCADE,
    FOREIGN KEY (id_variacao) REFERENCES Produtos_Variacoes(Id)
) ENGINE = InnoDB;

CREATE TABLE pagamentos_venda (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_venda INT NOT NULL,
    tipo_pagamento VARCHAR(50), 
    valor DECIMAL(10,2),
    FOREIGN KEY (id_venda) REFERENCES Vendas(id) ON DELETE CASCADE
) ENGINE = InnoDB;

CREATE TABLE log_sistema (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome_usuario VARCHAR(100) NOT NULL,
    acao VARCHAR(150) NOT NULL,
    formulario VARCHAR(100),
    data_hora DATETIME,
    INDEX IX_UserLogs_Data (data_hora DESC)
) ENGINE = InnoDB;

-- ==================================================================================
-- 5. PEDIDOS DE COMPRA
-- ==================================================================================

CREATE TABLE PedidosCompra (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    FornecedorId INT NOT NULL,
    DataPedido DATETIME DEFAULT CURRENT_TIMESTAMP,
    Status VARCHAR(20) DEFAULT 'Pendente',
    ValorTotal DECIMAL(10,2),
    ValorFrete DECIMAL(10,2) DEFAULT 0.00,
    PrazoEntregaDias INT DEFAULT 0,
    Observacoes TEXT,
    FOREIGN KEY (FornecedorId) REFERENCES Fornecedores(Id)
) ENGINE = InnoDB;

CREATE TABLE ItensPedidoCompra (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    PedidoId INT NOT NULL,
    VariacaoId INT NOT NULL, 
    Quantidade INT NOT NULL,
    CustoUnitario DECIMAL(10,2) NOT NULL,
    NomeProdutoSnapshot VARCHAR(200), -- Opcional: Para guardar o nome histórico
    FOREIGN KEY (PedidoId) REFERENCES PedidosCompra(Id) ON DELETE CASCADE,
    FOREIGN KEY (VariacaoId) REFERENCES Produtos_Variacoes(Id)
) ENGINE = InnoDB;

SET FOREIGN_KEY_CHECKS = 1;

-- 6. VIEWS (VERSÕES FINAIS CORRIGIDAS)
-- ==================================================================================

-- 6.1 View Estoque Geral
CREATE OR REPLACE VIEW vw_EstoqueGeral AS
SELECT 
    pv.Id AS IdVariacao,
    p.Id AS IdProdutoPai,
    p.Nome AS Descricao,
    pv.CodigoBarrasEAN,
    p.Categoria,
    p.SubCategoria,
    pv.Tamanho,
    p.Cor, 
    pv.QtdMin,
    pv.QtdMax,
    pv.QtdAtual
FROM Produtos_Variacoes pv
INNER JOIN Produtos p ON pv.ProdutoId = p.Id;

-- 6.2 View Cotações (Inteligência de Compra)
CREATE OR REPLACE VIEW vw_CotacoesPorVariacao AS
SELECT 
    pv.Id AS IdVariacao,
    p.Id AS IdProdutoPai,
    CONCAT(p.Nome, ' - ', pv.Tamanho) AS DescricaoCompleta,
    f.Id AS IdFornecedor,
    f.NomeFantasia AS NomeFornecedor,
    pfc.PrecoCustoTabela,
    IFNULL(fc.PrazoEntregaDias, 7) AS PrazoEntregaDias,
    IFNULL(fc.TaxaFretePadrao, 0) AS TaxaFretePadrao
FROM Produtos_Variacoes pv
INNER JOIN Produtos p ON pv.ProdutoId = p.Id
INNER JOIN Produtos_Fornecedores_Catalogo pfc ON p.Id = pfc.ProdutoPaiId
INNER JOIN Fornecedores f ON pfc.FornecedorId = f.Id
LEFT JOIN Fornecedores_Condicoes fc ON f.Id = fc.FornecedorId;

-- 6.3 View Detalhada (Usada no Front-End para preencher Grids)
CREATE OR REPLACE VIEW vw_ProdutosDetalhados AS
SELECT 
    pv.Id AS IdVariacao,
    CONCAT(p.Nome, ' - ', pv.Tamanho) AS NomeProduto,
    IFNULL(pv.QtdAtual, 0) AS EstoqueAtual,
    IFNULL(pv.QtdMin, 0) AS EstoqueMinimo,
    IFNULL(pv.QtdMax, 0) AS EstoqueMaximo
FROM Produtos_Variacoes pv
INNER JOIN Produtos p ON pv.ProdutoId = p.Id;

    pc.Id AS PedidoId,

    pc.DataPedido,
    pc.ValorTotal,
    pc.Status
FROM PedidosCompra pc

WHERE pc.Status = 'Pendente';
